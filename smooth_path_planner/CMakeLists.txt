cmake_minimum_required(VERSION 3.8)
project(smooth_path_planner)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
add_library(astar src/astar.cpp)
target_include_directories(astar PUBLIC include)
add_executable(planner_node src/planner_node.cpp)
target_link_libraries(planner_node astar)
ament_target_dependencies(planner_node rclcpp nav_msgs geometry_msgs)
install(TARGETS planner_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config maps rviz DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_astar test/test_astar.cpp)
  target_link_libraries(test_astar astar)
  target_compile_options(test_astar PRIVATE -UNDEBUG)
  add_test(NAME astar_behavior COMMAND test_astar)
endif()
ament_package()
